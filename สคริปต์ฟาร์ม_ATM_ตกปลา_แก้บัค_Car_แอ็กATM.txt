local SafeZoneFilterList = {}
local function DisableSafeZoneIndicators()
    local map = workspace:FindFirstChild("Map")
    if not map then return end
    local szi = map:FindFirstChild("SafeZoneIndicators")
    if not szi then return end
    SafeZoneFilterList = {}
    for _, sz in ipairs(szi:GetChildren()) do
        table.insert(SafeZoneFilterList, sz)
        for _, part in ipairs(sz:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
                part.CanTouch = false
                part.CanQuery = false
                table.insert(SafeZoneFilterList, part)
            end
        end
    end
end

    local filter = {Character}
    for _, obj in ipairs(SafeZoneFilterList) do table.insert(filter, obj) end
    if extra then for _, obj in ipairs(extra) do table.insert(filter, obj) end end
    return filter
end

local lastClick = 0
local aw_conn = nil
local function inTarget(n)
    local tp = SliderMinigame.target_pos.get()
    local ts = SliderMinigame.target_size.get()
    local len = SliderConstants.LENIENCE or 0.05
    if not tp or tp == 0 then return false end
    local h = ts / 2
    return n >= tp - h - len and n <= tp + h + len
end
local function fire()
    local s = getgenv().SliderSession
    if not s or not s.id or not s.active then return end
    if os.clock() - lastClick < 1.5 then return end
    local lvl = SliderMinigame.level.get()
    local pos = SliderMinigame.needle_pos.get()
    local t = workspace:GetServerTimeNow()
    pcall(function() Net.send("minigame_click", s.id, lvl, pos, t, s.version or 1) end)
    lastClick = os.clock()
end
local function startAW()
    if aw_conn then return end
    aw_conn = RunService.RenderStepped:Connect(function()
        local s = getgenv().SliderSession
        if not s or not s.active then return end
        if SliderMinigame.level.get() == 0 then return end
        if inTarget(SliderMinigame.needle_pos.get()) then fire() end
    end)
end
local function stopAW()
    if aw_conn then aw_conn:Disconnect() aw_conn = nil end
end
pcall(function()
    SliderMinigame.enabled.hook(function(v)
        if v then startAW() else stopAW() end
    end)
end)
startAW()

local function CreateJumpCircle(position)
    local circle = Instance.new("Part")
    circle.Name = "JumpCircle"
    circle.Shape = Enum.PartType.Cylinder
    circle.Size = Vector3.new(0.3, 7, 7)
    circle.Anchored = true
    circle.CanCollide = false
    circle.CanTouch = false
    circle.CanQuery = false
    circle.Material = Enum.Material.Neon
    circle.BrickColor = BrickColor.new("Lime green")
    circle.Transparency = 0.3
    circle.CFrame = CFrame.new(position) * CFrame.Angles(0, 0, math.rad(90))
    circle.Parent = workspace
    table.insert(jumpVisuals, circle)
    
    local innerCircle = Instance.new("Part")
    innerCircle.Name = "JumpCircleInner"
    innerCircle.Shape = Enum.PartType.Cylinder
    innerCircle.Size = Vector3.new(0.4, 3, 3)
    innerCircle.Anchored = true
    innerCircle.CanCollide = false
    innerCircle.CanTouch = false
    innerCircle.CanQuery = false
    innerCircle.Material = Enum.Material.Neon
    innerCircle.BrickColor = BrickColor.new("Cyan")
    innerCircle.Transparency = 0.5
    innerCircle.CFrame = CFrame.new(position + Vector3.new(0, 0.3, 0)) * CFrame.Angles(0, 0, math.rad(90))
    innerCircle.Parent = workspace
    table.insert(jumpVisuals, innerCircle)
    
    task.spawn(function()
        for i = 1, 8 do
            if not circle.Parent then break end
            local size = 7 + i * 1.5
            circle.Size = Vector3.new(0.3, size, size)
            circle.Transparency = 0.3 + (i * 0.08)
            innerCircle.Size = Vector3.new(0.4, 3 + i * 0.5, 3 + i * 0.5)
            innerCircle.Transparency = 0.5 + (i * 0.06)
            task.wait(0.08)
        end
        if circle.Parent then circle:Destroy() end
        if innerCircle.Parent then innerCircle:Destroy() end
    end)
    return circle
end

local function DrawPath(waypoints)
    ClearPathLines()
    if not waypoints or #waypoints < 2 then return end
    for i = 1, #waypoints - 1 do
        local p1 = waypoints[i].Position
        local p2 = waypoints[i+1].Position
        local dist = (p2 - p1).Magnitude
        if dist > 0.5 then
            local mid = (p1 + p2) / 2
            local dir = (p2 - p1).Unit
            local part = Instance.new("Part")
            part.Size = Vector3.new(0.4, 0.4, dist)
            part.CFrame = CFrame.lookAt(mid, mid + dir)
            part.Anchored = true
            part.CanCollide = false
            part.Material = Enum.Material.Neon
            part.BrickColor = BrickColor.new("Bright blue")
            part.Transparency = 0.25
            part.Parent = workspace
            table.insert(pathParts, part)
        end
    end
end

function Sf:LockOwnVehicle(force)
    local now = tick()
    if not force and (now - LastVehicleLockTime) < VEHICLE_LOCK_INTERVAL then
        return
    end
    for _, vehicle in pairs(workspace.Vehicles:GetChildren()) do
        if vehicle:IsA("Model") and vehicle:GetAttribute("OwnerUserId") == UserId then
            local locked = vehicle:GetAttribute("IsLocked")
            if locked ~= true then
                print("[Lock] Sperre Fahrzeug:", vehicle.Name)
                pcall(function() Net.send("lock_vehicle", vehicle, 2) end)
                LastVehicleLockTime = now
            else
                LastVehicleLockTime = now
            end
            return
        end
    end
end

function Sf:ResolveRodName()

    if level >= 60 then return "FishingRodUltimate"
    elseif level >= 35 then return "FishingRodAdvanced"
    elseif level >= 20 then return "FishingRodPro"
    else return "FishingRodRegular" end
end

    elseif level >= 25 then return "PrawntecRegular"

end

function Sf:CountBaits()
    local total = 0
    local Items = PlayerGui:FindFirstChild("Items")
    if Items then
        local holder = Items:FindFirstChild("ItemsHolder")
        if holder then
            local Sc = holder:FindFirstChild("ItemsScrollingFrame")
            if Sc then
                for _, v in pairs(Sc:GetChildren()) do
                    if v.Name ~= 'Folder' and v.Name ~= 'UIGridLayout' and v.Name ~= "ItemTemplate" then
                        local itemName = v:FindFirstChild("ItemName")
                        if itemName and itemName.Text then
                            local t = string.lower(itemName.Text)
                            if string.find(t, "wormtec") or string.find(t, "prawntec") then
                                local destroyedIcon = v:FindFirstChild("DestroyedItemIcon")
                                local isDestroyed = destroyedIcon and destroyedIcon.Visible
                                local isVisible = v.Visible
                                
                                if not isDestroyed and isVisible then
                                    local countLabel = v:FindFirstChild("ItemCount")
                                    if countLabel and countLabel:IsA("TextLabel") then
                                        local num = tonumber(countLabel.Text:match("%d+"))
                                        if num and num > 0 then
                                            total = total + num
                                        end
                                    else
                                        total = total + 1
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    local hotbar = PlayerGui:FindFirstChild("Hotbar")
    if hotbar then
        for _, frame in ipairs(hotbar:GetDescendants()) do
            if frame.Name == "Ammo" and frame:IsA("TextLabel") then
                local num = tonumber(frame.Text:match("(%d+)/"))
                if num and num > 0 then
                    total = total + num
                end
            end
        end
    end
    return total
end

                                end
                            else
                                count = count + 1
                            end
                        end
                    end
                end
            end
        end
    end
    return count
end

function Sf:WalkTo(targetPos, timeout)
    if not targetPos then return false end
    if not Character or not Character.Parent then
        Character = Client.Character or Client.CharacterAdded:Wait()
        Humanoid = Character:WaitForChild("Humanoid")
        RootPart = Character:WaitForChild("HumanoidRootPart")
    end
    Config.StopWalking = false
    Config.Running = true
    local Root = Character:WaitForChild("HumanoidRootPart")
    local Hum = Character:WaitForChild("Humanoid")

    local path = PathfindingService:CreatePath({
        AgentCanJump = true, AgentJumpHeight = 7.2, AgentHeight = 5,
        AgentRadius = 2, AgentMaxSlope = 50, WaypointSpacing = 4,
        Costs = {BlockedNode = 100, DoorArea = 1}
    })
    local ok = pcall(function() path:ComputeAsync(Root.Position, targetPos) end)

    if not ok or path.Status ~= Enum.PathStatus.Success then
        warn("[Walk] Pathfinder Error:", tostring(path.Status))
        Config.Running = false
        ClearPathLines()
        Sf:KillSelf()
        return false
    end

    local waypoints = path:GetWaypoints()
    DrawPath(waypoints)

    for _, wp in ipairs(waypoints) do
        if shouldStop() then break end
        local target = wp.Position + Vector3.new(0, 3, 0)
        local startTime = tick()
        while (Root.Position - target).Magnitude > 1.5 do
            if shouldStop() then break end
            if tick() - startTime > 10 then break end
            local currentSpeed = Config.WalkingSpeed
            task.wait(0.03)
            if shouldStop() then break end
            local curPos = Root.Position
            local delta = target - curPos
            local remaining = delta.Magnitude
            local step = math.min(currentSpeed * 0.03, remaining)
            Root:PivotTo(CFrame.new(curPos + delta.Unit * step))
            Net.send("set_sprinting_1", true)
        end

        if wp.Action == Enum.PathWaypointAction.Jump then
            local jumpPos = Vector3.new(wp.Position.X, wp.Position.Y, wp.Position.Z)
            CreateJumpCircle(jumpPos)
            task.wait(0.1)
            Hum:ChangeState(Enum.HumanoidStateType.Jumping)
            task.wait(0.15)
            if Hum.FloorMaterial == Enum.Material.Air then
                Hum.Jump = true
            end
            local landTimeout = tick() + 1.5
            while Hum.FloorMaterial == Enum.Material.Air and tick() < landTimeout do
                task.wait(0.05)
            end
            task.wait(0.1)
        end
        if shouldStop() then break end
    end

    Config.Running = false
    ClearPathLines()
    task.delay(3, ClearJumpVisuals)
    return not shouldStop()
end

function Sf:EnterVehicle(vehicle, timeout)
    if not vehicle or not vehicle.PrimaryPart then return false end
    timeout = timeout or 8
    local st = tick()
    while not Humanoid.Sit and tick() - st < timeout do
        if shouldStop() then return false end
        pcall(function()
            for _, prompt in pairs(vehicle.PrimaryPart:GetDescendants()) do
                if prompt:IsA('ProximityPrompt') then
                    fireproximityprompt(prompt)
                    task.wait(0.15)
                end
            end
        end)
        task.wait(0.2)
    end
    if Humanoid.Sit then
        task.wait(0.3)
        Sf:LockOwnVehicle(true)
    end
    return Humanoid.Sit
end

function Sf:ExitVehicle(timeout)
    if not Humanoid.Sit then return true end
    pcall(function() Net.get("exit_seat") end)
    local st = tick()
    while Humanoid.Sit and tick() - st < (timeout or 4) do
        task.wait(0.15)
    end
    return not Humanoid.Sit
end

function Sf:DriveVehicleUnderground(model, destination, value, customSpeed, onArrive, customVehicleType, stayFlying)
    if not model or not model.PrimaryPart then return false end
    
    local underground = Config.FishingUnderground
    local undergroundDepth = math.max(10, tonumber(Config.FarmUndergroundDepth) or 55)
    
    if underground then
        local surfaceDestination = destination
        local travelDestination = destination + Vector3.new(0, -undergroundDepth, 0)
        
        local pivot = model:GetPivot()
        model:PivotTo(CFrame.new(pivot.Position.X, travelDestination.Y, pivot.Position.Z) * pivot.Rotation)
        task.wait(0.05)
        
        local startPos = model.PrimaryPart.Position
        local delta = travelDestination - startPos
        local dist = delta.Magnitude
        
        if dist > 0.05 then
            local dir = delta.Unit
            local speed = math.max(1, tonumber(customSpeed) or (Config.FishingTravelType == "Car" and Config.CarSpeed or Config.BikeSpeed))
            local started = tick()
            
            while shouldContinue(value) and model and model.PrimaryPart do
                task.wait()
                if not Humanoid.Sit then
                    if not Sf:EnterVehicle(model, 3) then break end
                end
                
                local moved = math.min((tick() - started) * speed, dist)
                local newPos = startPos + dir * moved
                local rotation = CFrame.lookAt(newPos, newPos + dir)
                
                local vehicleTypeForWheelie = customVehicleType or Config.FishingTravelType
                if Config.WheelieEnabled and (not Config.WheelieOnlyBike or vehicleTypeForWheelie == 'Bike') then
                    rotation = rotation * CFrame.Angles(math.rad(math.clamp(Config.WheelieAngle, 0, 35)), 0, 0)
                end
                
                model:PivotTo(rotation)
                for _, part in pairs(model:GetDescendants()) do
                    if part:IsA('BasePart') then
                        part.AssemblyLinearVelocity = Vector3.zero
                        part.AssemblyAngularVelocity = Vector3.zero
                    end
                end
                
                if moved >= dist then break end
            end
        end
        
        if model and model.PrimaryPart then
            if stayFlying then
                local freezePos = Vector3.new(destination.X, destination.Y, destination.Z)
                model:PivotTo(CFrame.new(freezePos) * model:GetPivot().Rotation)
                for _, part in pairs(model:GetDescendants()) do
                    if part:IsA('BasePart') then
                        part.Anchored = true
                        part.AssemblyLinearVelocity = Vector3.zero
                        part.AssemblyAngularVelocity = Vector3.zero
                    end
                end
                ActiveFishingVehicle = model
                print("[Freeze] Fahrzeug steht auf Angelposition Y="..freezePos.Y)
            else
                local restorePos
                if shouldContinue(value) then
                    restorePos = surfaceDestination
                else
                    local p = model.PrimaryPart.Position
                    restorePos = Vector3.new(p.X, surfaceDestination.Y, p.Z)
                end
                model:PivotTo(CFrame.new(restorePos) * model:GetPivot().Rotation)
                for _, part in pairs(model:GetDescendants()) do
                    if part:IsA('BasePart') then
                        part.AssemblyLinearVelocity = Vector3.zero
                        part.AssemblyAngularVelocity = Vector3.zero
                        part.Anchored = false
                    end
                end
            end
            if onArrive then
                pcall(onArrive)
            end
        end
        Config.Running = false
        ClearPathLines()
        return true
    end
    
    return false
end

function Sf:PullVehicle()
    local root = Character and Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    for _, vehicle in pairs(workspace.Vehicles:GetChildren()) do
        if vehicle:IsA("Model") and vehicle:GetAttribute("OwnerUserId") == UserId then
            local pp = vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart")
            if not pp then return end
            local pos = root.CFrame * CFrame.new(0, 3, -12)
            local rp = RaycastParams.new()
            rp.FilterDescendantsInstances = getRaycastFilter({vehicle})
            rp.FilterType = Enum.RaycastFilterType.Exclude
            local ray = workspace:Raycast(pos.Position, Vector3.new(0, -200, 0), rp)
            local groundY = ray and ray.Position.Y or 260
            local offsetY = pp.Size.Y / 2
            if ActiveFishingVehicle then
                for _, part in pairs(ActiveFishingVehicle:GetDescendants()) do
                    if part:IsA("BasePart") then part.Anchored = false end
                end
                ActiveFishingVehicle = nil
            end
            pcall(function() vehicle:SetPrimaryPartCFrame(CFrame.new(pos.X, groundY + offsetY, pos.Z) * vehicle:GetPivot().Rotation) end)
            for _, part in pairs(vehicle:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.AssemblyLinearVelocity = Vector3.zero
                    part.AssemblyAngularVelocity = Vector3.zero
                    part.Anchored = false
                end
            end
        end
    end
end

local function GetOrSpawnVehicle(vehicleType)
    if vehicleType == "Walk" then return nil end
    for _, v in pairs(workspace.Vehicles:GetChildren()) do
        if v:GetAttribute('OwnerUserId') == UserId then return v end
    end
    local vName = Sf:GetCarFromType(vehicleType)
    if not vName then
        if vehicleType == "Bike" then Sf:BuyBikeCrate() else Sf:BuyCarCrate() end
        task.wait(1.5)
        return nil
    end
    local Checker = Sf:GetInfo(vName)
    if Checker[2] and not Checker[4] and not Checker[5] then
        Net.get("toggle_equip_item", tostring(Checker[3]))
        task.wait(0.2)
    end
    task.wait(0.5)
    for _, v in pairs(workspace.Vehicles:GetChildren()) do
        if v:GetAttribute('OwnerUserId') == UserId then return v end
    end
    return nil
end

function Sf:GoToPosition(targetPos, timeout)
    if Humanoid.Sit then
        Sf:ExitVehicle(5)
        task.wait(0.3)
    end
    BusyWalk = true
    local attempts = 0
    local reached = false
    while distTo(targetPos) > 4 and attempts < 3 do
        if shouldStop() then break end
        local success = Sf:WalkTo(targetPos, timeout or 30)
        if not success then
            BusyWalk = false
            return false
        end
        attempts = attempts + 1
        task.wait(0.1)
    end
    BusyWalk = false
    reached = distTo(targetPos) <= 8
    return reached
end

function Sf:TravelTo(destination, value, stayFlying)
    if not destination then return false end
    if not Config.AutoFarmFishing and not BusyWalk then return false end
    local dist = distTo(destination)
    
    if Config.FishingTravelType == "Walk" or dist <= 10 then
        if Humanoid.Sit then
            Sf:ExitVehicle(5)
            task.wait(0.3)
        end
        return Sf:WalkTo(destination, 40)
    end
    
    if not Config.AutoFarmFishing and not BusyWalk then return false end
    
    local vehicle = GetOrSpawnVehicle(Config.FishingTravelType)
    if vehicle then
        if not Humanoid.Sit then
            Sf:PullVehicle()
            task.wait(0.4)
            if not Config.AutoFarmFishing then return false end
            if not Sf:EnterVehicle(vehicle, 6) then
                return Sf:WalkTo(destination, 40)
            end
        end
        
        local speed = (Config.FishingTravelType == "Car") and Config.CarSpeed or Config.BikeSpeed
        local success = Sf:DriveVehicleUnderground(vehicle, destination, value, speed, nil, Config.FishingTravelType, stayFlying)
        return success
    else
        return Sf:WalkTo(destination, 40)
    end
end

function FishingLoop()
    if CountFish() > 0 then
        SellAllFish()
        if not Config.AutoFarmFishing then fullCleanup(true) return end
    end

    while Config.AutoFarmFishing do
        if not Config.AutoFarmFishing then break end

        if Sf:HandleDeathScreen() then
            task.wait(1)
            continue
        end

        pcall(function() Sf:LockOwnVehicle(false) end)

        pcall(function()
            Status.Fish = CountFish()
            Status.Money = Sf:GetMoney()
            Status.Bank = Sf:ATMMoney()
            Status.Baits = Sf:CountBaits()
        end)

        local hasRod, rodUid, actualRodName, rodEquipped = false, nil, nil, false
        local hasBait, baitUid = false, nil
        local Items = PlayerGui:FindFirstChild("Items")
        if Items then
            local H = Items:FindFirstChild("ItemsHolder") and Items.ItemsHolder:FindFirstChild("ItemsScrollingFrame")
            if H then
                for _, v in pairs(H:GetChildren()) do
                    if not Config.AutoFarmFishing then break end
                    if v.Name ~= 'Folder' and v.Name ~= 'UIGridLayout' and v.Name ~= "ItemTemplate" then
                        local itemName = v:FindFirstChild("ItemName")
                        if itemName and itemName.Text then
                            local t = string.lower(itemName.Text)
                            if not hasRod and string.find(t, "rod") then
                                hasRod = true rodUid = v.Name actualRodName = itemName.Text
                                local eq = v:FindFirstChild("ItemEquipped") rodEquipped = eq and eq.Visible or false
                            end
                            if not hasBait and (string.find(t, "wormtec") or string.find(t, "prawntec")) then
                                hasBait = true baitUid = v.Name
                            end
                        end
                    end
                end
            end
        end

        if not Config.AutoFarmFishing then break end

        if not hasRod or not hasBait then
            if fishingBaseplate then fishingBaseplate:Destroy() fishingBaseplate = nil end
            releaseFlyingVehicle()
            if Humanoid.Sit then
                Sf:ExitVehicle(5)
                task.wait(0.3)
            end
            if not Config.AutoFarmFishing then break end
            local realRodName = Sf:ResolveRodName()
            local realBaitName = Sf:ResolveBaitName()
            local neededMoney = 0
            if not hasRod then for _, rod in ipairs(FishingRods) do if rod.Name == realRodName then neededMoney = neededMoney + rod.Price break end end end
            if not hasBait then for _, bait in ipairs(FishingBaits) do if bait.Name == realBaitName then neededMoney = neededMoney + (bait.Price * Config.FishingBaitAmount) break end end end
            local totalMoney = Sf:GetMoney() + Sf:ATMMoney()
            if totalMoney < neededMoney then
                task.wait(5)
                continue
            end
            while Sf:GetMoney() < neededMoney and Config.AutoFarmFishing do
                if Humanoid.Sit then Sf:ExitVehicle(5) task.wait(0.3) end
                Sf:TravelTo(POS.ATM, Config.AutoFarmFishing, false)
                if not Config.AutoFarmFishing then break end
                local need = neededMoney - Sf:GetMoney()
                if need > 0 and Sf:ATMMoney() >= need then
                    Net.get("transfer_funds", "bank", "hand", need)
                    task.wait(0.3)
                else break end
            end
            if not Config.AutoFarmFishing then break end
            if Humanoid.Sit then Sf:ExitVehicle(5) task.wait(0.3) end
            Sf:TravelTo(POS.Shop, Config.AutoFarmFishing, false)
            if not Config.AutoFarmFishing then break end
            if Humanoid.Sit then Sf:ExitVehicle(5) task.wait(0.3) end
            if not hasRod then 
                Net.get("purchase_consumable", workspace:WaitForChild("ShopZone_Hardware"), realRodName) 
                task.wait(0.15) 
            end
            if not Config.AutoFarmFishing then break end
            if not hasBait then
                for i = 1, Config.FishingBaitAmount do
                    if not Config.AutoFarmFishing then break end
                    Net.get("purchase_consumable", workspace:WaitForChild("ShopZone_Hardware"), realBaitName)
                    task.wait(0.05)
                end
            end
            task.wait(0.5)
            continue
        end

        local baitCount = Sf:CountBaits()
        if baitCount <= 0 then
            releaseFlyingVehicle()
            if Humanoid.Sit then Sf:ExitVehicle(5) task.wait(0.3) end
            if CountFish() > 0 then
                SellAllFish()
                if not Config.AutoFarmFishing then break end
            end
            BuyBaits(Config.FishingBaitAmount)
            if not Config.AutoFarmFishing then break end
            continue
        end

        local level = Sf:GetLevel()
        local fishingSpot, throwPos, requiredLevel = Sf:GetFishingSpot()
        Status.State = "Traveling (Lvl " .. requiredLevel .. ")"
        
        local useFlying = (Config.FishingTravelType ~= "Walk")
        local travelSuccess = Sf:TravelTo(fishingSpot, Config.AutoFarmFishing, useFlying)
        if not Config.AutoFarmFishing then break end
        if not travelSuccess then 
            if fishingBaseplate then fishingBaseplate:Destroy() fishingBaseplate = nil end
            continue 
        end

        if useFlying then
            print("Done - Fahrzeug steht auf Angelposition")
            task.wait(0.5)
        else
            if Humanoid.Sit then Sf:ExitVehicle(4) task.wait(0.2) end
            if not Config.AutoFarmFishing then break end
            if fishingBaseplate and fishingBaseplate.Parent then fishingBaseplate:Destroy() end
            
            if Config.FishingTravelType == "Walk" and level < 10 then
                RootPart.CFrame = CFrame.new(RootPart.Position.X, 260, RootPart.Position.Z)
                task.wait(0.4)
                RootPart.CFrame = CFrame.new(POS.FishLowLevelWalkDrop)
                task.wait(0.4)
            else
                RootPart.CFrame = RootPart.CFrame + Vector3.new(0, 3, 0)
                task.wait(0.1)
            end
            
            fishingBaseplate = Instance.new("Part")
            fishingBaseplate.Name = "FishingBaseplate"
            fishingBaseplate.Size = Vector3.new(10, 1, 10)
            fishingBaseplate.Anchored = true
            fishingBaseplate.CanCollide = true
            fishingBaseplate.Position = RootPart.Position - Vector3.new(0, 3, 0)
            fishingBaseplate.Parent = workspace
            task.wait(0.1)
            RootPart.Anchored = true
            print("Done - Baseplate")
            task.wait(0.5)
        end

        if not Config.AutoFarmFishing then break end

        if not rodEquipped then
            Net.get("toggle_equip_item", tostring(rodUid))
            task.wait(0.3)
        end
        
        local rodInHand = Sf:EnsureRodEquipped(actualRodName)
        if not rodInHand then
            task.wait(1)
            rodInHand = Sf:EnsureRodEquipped(actualRodName)
        end
        
        Net.get("equip_ammo_on_item", rodUid, baitUid)
        task.wait(0.3)

        Status.State = "Fishing (Lvl " .. requiredLevel .. ")"
        local maxIterations = 500
        local iteration = 0
        while Config.AutoFarmFishing and CountFish() < 15 and iteration < maxIterations do
            iteration = iteration + 1
            if not Config.AutoFarmFishing then break end
            Status.Fish = CountFish()
            Status.Baits = Sf:CountBaits()

            if Sf:HandleDeathScreen() then
                task.wait(1)
                break
            end

            if Sf:CountBaits() <= 0 then
                break
            end

            if IsMinigameActive() then
                task.wait(0.3)
                continue
            end

            local char = Client.Character
            if not char then break end
            local rodTool = char:FindFirstChild(actualRodName)
            if not rodTool then
                rodTool = Sf:EnsureRodEquipped(actualRodName)
                if not rodTool then
                    task.wait(0.5)
                    continue
                end
            end

            if rodTool then
                if IsMinigameActive() then
                    task.wait(0.3)
                    continue
                end

                local throwOk = pcall(function()
                    Net.get("throw_rod", rodTool, throwPos)
                end)
                if not throwOk then task.wait(0.5) continue end

                local timeout = tick() + 25
                local minigameStarted = false
                while Config.AutoFarmFishing and tick() < timeout do
                    task.wait(0.05)
                    if IsMinigameActive() then
                        minigameStarted = true
                        break
                    end
                end

                if minigameStarted then
                    while Config.AutoFarmFishing and IsMinigameActive() do
                        task.wait(0.1)
                    end
                    task.wait(0.05)
                    pcall(function() Net.send("reel_ended", rodTool, true) end)
                    task.wait(0.3)
                end

                if Sf:CountBaits() > 0 and rodUid and baitUid then
                    Net.get("equip_ammo_on_item", rodUid, baitUid)
                    task.wait(0.15)
                end

                task.wait(0.3)
            else
                task.wait(1)
            end
        end

        if not Config.AutoFarmFishing then break end

        local currentFish = CountFish()
        local shouldSell = currentFish >= 15 or Sf:CountBaits() <= 0
        if shouldSell and currentFish > 0 then
            SellAllFish()
            if not Config.AutoFarmFishing then break end
        end

        if Sf:CountBaits() <= 0 then
            BuyBaits(Config.FishingBaitAmount)
            if not Config.AutoFarmFishing then break end
        end
        Status.State = "Idle"
    end
    fullCleanup(true)
end
